\section{Setup}
\begin{frame}{Setup}
\begin{itemize}
    \item Expand the provided PythonCourse.zip file to a location of your choice.
    \item Cmd prompt at the PythonCourse folder (right click on folder, select "Open with cmd"). For macOS, open terminal and \texttt{cd} to the PythonCourse folder.
    \item type: "\texttt{python --version <Enter>}" (or try \texttt{python3 --version <Enter>} on macOS), you should see the Python version you installed. If you don't, ask AI until you do (reinstall python for example).
    \item type: "\texttt{python setup\_env.py <Enter>}"
    \item At the end of the script, it says "Next steps: Activate the environment" and tells you what to type. Type that.
    \item type: "\texttt{jupyter notebook <Enter>}". This should open a browser window with the Jupyter interface. This is the basic setup, we can work from here, but I am showing you a better way too. 
\end{itemize}
\end{frame}

\begin{frame}{Better Way}
\begin{itemize}
    \item Open VSCode
    \item Open folder: \textbf{File $\rightarrow$ Open Folder...} and select the unzipped PythonCourse folder.
    \item On the left side you find the file explorer.  Click \texttt{python\_intro\_stats\_notebook.ipynb}
    \item If VS Code asks to install recommended extensions, click \textbf{Yes}.
    \item If VS Code asks what kernel to use, always choose "python virtual environment" and click the one called "venv".
    \item If VS Code complains about "could not find python interpreter for jupytext", just click away the popup.
\end{itemize}
\end{frame}

\section{GitHub Collaboration}

\begin{frame}{Collaborating on LaTeX via GitHub}
\begin{itemize}
    \item \textbf{Initialize repository}: Create a repo on GitHub, clone it locally
    \item \textbf{Create a branch} for your feature: \texttt{git checkout -b feature-name}
    \item \textbf{Edit your .tex file(s)} and commit regularly:
    \begin{itemize}
        \item \texttt{git add <file>}
        \item \texttt{git commit -m "Clear message about changes"}
        \item \texttt{git push origin feature-name}
    \end{itemize}
    \item \textbf{Open a Pull Request (PR)} on GitHub for review
    \item \textbf{Resolve conflicts} if others edited the same lines
    \item \textbf{Merge} after approval
\end{itemize}
\end{frame}

\begin{frame}{Tips for LaTeX Collaboration}
\begin{itemize}
    \item \textbf{One sentence per line} in your .tex files—makes diffs clearer and conflicts easier to resolve
    \item \textbf{Commit frequently}—small, logical changes are easier to review and merge
    \item \textbf{.gitignore}—exclude build artifacts:
    \begin{itemize}
        \item \texttt{*.aux}, \texttt{*.log}, \texttt{*.pdf}, \texttt{*.fdb\_latexmk}, \texttt{*.fls}, \texttt{*.nav}, \texttt{*.snm}, \texttt{*.toc}
    \end{itemize}
    \item \textbf{Communicate}—discuss large structural changes in issues before editing
    \item \textbf{Use descriptive branch names}: \texttt{fix-typos}, \texttt{add-appendix}, \texttt{reformat-intro}
\end{itemize}
\end{frame}
